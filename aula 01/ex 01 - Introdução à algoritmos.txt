VAR
	REAL: baseTriangulo, alturaTriangulo, areaTriangulo;

	ESCREVA("Informe a altura do triangulo: ");
	LEIA(alturaTriangulo);

	areaTriangulo <- (baseTriangulo * altura) / 2;

ALGORITMO tempoEmSegundos
VAR
	inteiro: segundos, minutos, horas, totalSegundos;

	ESCREVA("Informe os minutos: ");
	LEIA(minutos);

	ESCREVA("Informe os segundos: ");

7. Escreva um algoritmo que calcule e mostre o consumo médio e a autonomia que um veículo ainda teria antes de um abastecimento de combustível. Considere que o veículo sempre seja abastecido até encher o tanque e que são fornecidas apenas a capacidade do tanque, a quantidade de litros abastecidos e a quilometragem percorrida desde o último abastecimento.

ALGORITMO mediaVeiculo
VAR
	REAL: capacidadeTanque, litrosAbastecidos, quilometragemPercorrida, consumoMedio, autonomia;
INICIO
	ESCREVA("Informe quantos litros o tanque comporta: ");
	LEIA(capacidadeTanque);

	ESCREVA("Informe o valor em litros que foi abastecido:");
	LEIA(litrosAbastecidos);

	ESCREVA("Informe a quilometragem que foi percorrida: ");
	LEIA(quilometragemPercorrida);

	consumoMedio <- quilometragemPercorrida/litrosAbastecidos;
	litrosRestantes <- capacidadeTanque – litrosAbastecidos;
	autonomia <- consumoMedio * litrosRestantes;
	
	ESCREVA("O consumo médio foi de : " , consumoMedio);
	ESCREVA("O carro ainda pode percorrer : " , autonomia);
FIM
